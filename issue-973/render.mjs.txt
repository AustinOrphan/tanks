// Render card.html at 1200x630 CSS px, deviceScaleFactor 2, to a 2400x1260 PNG.
// Usage: node render.mjs <card.html> <out-2x.png> [extra-css]
// extra-css (optional) is injected after load, for scratch previews of variants only.
// Adapted from options/c-split-card/render.mjs: it also waits for the page's window.__laidOut flag,
// which card.html sets once the bundled face has loaded.
import { createRequire } from 'node:module';
import { resolve } from 'node:path';
import { pathToFileURL } from 'node:url';

const require = createRequire(
  '/Users/austinorphan/src/tanks/.claude/worktrees/imperative-waddling-willow/package.json',
);
const { chromium } = require('playwright');

const [htmlPath, outPath, extraCss] = process.argv.slice(2);
if (!htmlPath || !outPath) {
  console.error('usage: node render.mjs <card.html> <out-2x.png> [extra-css]');
  process.exit(2);
}

const browser = await chromium.launch();
try {
  const page = await browser.newPage({
    viewport: { width: 1200, height: 630 },
    deviceScaleFactor: 2,
  });
  const failed = [];
  page.on('requestfailed', (r) => failed.push(r.url()));
  page.on('pageerror', (e) => failed.push(`pageerror: ${e.message}`));
  await page.goto(pathToFileURL(resolve(htmlPath)).href, { waitUntil: 'load' });
  if (extraCss) await page.addStyleTag({ content: extraCss });
  await page.evaluate(() => document.fonts.ready);
  await page.waitForFunction(() => window.__laidOut === true, null, { timeout: 10000 });
  // Fail loudly if a bundled face did not load (a silent fallback would change the type).
  const fonts = await page.evaluate(() =>
    [...document.fonts].map((f) => `${f.family} ${f.weight} ${f.status}`),
  );
  console.log(fonts.join('\n'));
  if (fonts.some((f) => !f.endsWith('loaded'))) throw new Error('a font did not load');
  if (failed.length) throw new Error(`failed requests: ${failed.join(', ')}`);
  const metrics = await page.evaluate(() =>
    [...document.querySelectorAll('.copy > *')].map((el) => {
      const r = el.getBoundingClientRect();
      const label = `${el.className} "${el.textContent}"`;
      return `${label}: box x ${r.left.toFixed(1)}-${r.right.toFixed(1)}, y ${r.top.toFixed(1)}-${r.bottom.toFixed(1)}`;
    }),
  );
  console.log(metrics.join('\n'));
  await page.screenshot({ path: outPath, clip: { x: 0, y: 0, width: 1200, height: 630 } });
} finally {
  await browser.close();
}
